[HttpGet("{location}/frontpage")]
GetFrontPage(string location,
  int top = 10, int perSection = 0,
  string? section = null,
  int? excerpt = null,     // null = hel tekst
  int? heroExcerpt = null)
{
  var query = new FrontPageQuery(...);
  if (query.Validate() is {} error)
    return BadRequest(error);
  cached = _cache.GetFrontPageAsync(...);
  if (cached is not null) return Ok(cached);
  articles = _readRepository
    .GetFrontPageAsync(location, query);
  _cache.StoreFrontPageAsync(..., articles);
  return Ok(articles);
}

Efter (GetFrontPageAsync)
with ranked as (select ...,
  row_number() over (order by shown_at desc)
    as rn_all,
  row_number() over (partition by section_id
    order by shown_at desc) as rn_section
  from articles a join sections s ...),
picked as (select *,
  case when <hero> then coalesce(@HeroExcerpt,
    @Excerpt) else @Excerpt end as excerpt_length
  from ranked
  where rn_all <= @Top
     or rn_section <= @PerSection
     or (section = @Section and rn_section <= @Top))
select id, title, ...,
  case when excerpt_length is null then breadtext
       when excerpt_length = 0 then ''
       else left(breadtext, excerpt_length + 1)
  end as breadtext
from picked order by shown_at desc
